`timescale 1ns / 1ps

module circular_barrel_shifter #(
    parameter N = 8,                // Data width
    parameter S = $clog2(N)         // Shift amount width
)(
    input  wire [N-1:0] data_in,
    input  wire [S-1:0] shift_amt,
    input  wire         dir,        // 0 = Rotate Left, 1 = Rotate Right
    output reg  [N-1:0] data_out
);

    always @(*) begin
        // Bypass shifting if shift_amt is 0 to prevent (N-0) out-of-bounds warnings
        if (shift_amt == 0) begin
            data_out = data_in;
        end else if (dir == 1'b0) begin
            // Rotate Left: Shift left OR'ed with the bits falling off the top wrapped to the bottom
            data_out = (data_in << shift_amt) | (data_in >> (N - shift_amt));
        end else begin
            // Rotate Right: Shift right OR'ed with the bits falling off the bottom wrapped to the top
            data_out = (data_in >> shift_amt) | (data_in << (N - shift_amt));
        end
    end

endmodule
(note : not to be confused with << or >> as they pad in zeroes)
